# The generator (run as above):
#| # Independent reference for the SIMPLIFY.md S4 keys and member boxes
#| # (AUDIT.md G5: every derivation is keyed BLAKE2b; A6: k_g is hedged and
#| # eph_sk is derived, not drawn).
#| # BLAKE2b: hashlib. X25519: pyca/cryptography and the RFC 7748 §5 ladder
#| # written here in plain integers, which must agree. XChaCha20-Poly1305:
#| # pyca/cryptography's ChaCha20Poly1305 under an HChaCha20 subkey written
#| # here from draft-irtf-cfrg-xchacha-03, and a ChaCha20 + Poly1305 written
#| # here from RFC 8439, which must agree. No monocypher, no libsodium.
#| import hashlib, struct
#| from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey, X25519PublicKey
#| from cryptography.hazmat.primitives.ciphers.aead import ChaCha20Poly1305
#|
#| P = 2**255 - 19
#| def f(key, msg): return hashlib.blake2b(msg, key=key, digest_size=32).digest()
#| def u48(g): return struct.pack('>Q', g)[2:]
#|
#| def ladder(k, u):                       # RFC 7748 §5, X25519(k, u)
#|     k = bytearray(k); k[0] &= 248; k[31] &= 127; k[31] |= 64
#|     k = int.from_bytes(bytes(k), 'little')
#|     x1 = int.from_bytes(u, 'little') & ((1 << 255) - 1)
#|     x2, z2, x3, z3, swap = 1, 0, x1, 1, 0
#|     for t in range(254, -1, -1):
#|         kt = (k >> t) & 1; swap ^= kt
#|         if swap: x2, x3 = x3, x2; z2, z3 = z3, z2
#|         swap = kt
#|         a = (x2 + z2) % P; aa = a * a % P; b = (x2 - z2) % P; bb = b * b % P
#|         e = (aa - bb) % P; c = (x3 + z3) % P; d = (x3 - z3) % P
#|         da = d * a % P; cb = c * b % P
#|         x3 = pow(da + cb, 2, P); z3 = x1 * pow(da - cb, 2, P) % P
#|         x2 = aa * bb % P; z2 = e * (aa + 121665 * e) % P
#|     if swap: x2, x3 = x3, x2; z2, z3 = z3, z2
#|     return (x2 * pow(z2, P - 2, P) % P).to_bytes(32, 'little')
#|
#| def x25519(sk, pk):
#|     a = X25519PrivateKey.from_private_bytes(sk).exchange(X25519PublicKey.from_public_bytes(pk))
#|     b = ladder(sk, pk)
#|     assert a == b
#|     return a
#|
#| def x25519_pub(sk):
#|     a = X25519PrivateKey.from_private_bytes(sk).public_key().public_bytes_raw()
#|     b = ladder(sk, (9).to_bytes(32, 'little'))
#|     assert a == b
#|     return a
#|
#| def rotl(v, c): return ((v << c) & 0xffffffff) | (v >> (32 - c))
#| def qr(s, a, b, c, d):
#|     s[a] = (s[a] + s[b]) & 0xffffffff; s[d] = rotl(s[d] ^ s[a], 16)
#|     s[c] = (s[c] + s[d]) & 0xffffffff; s[b] = rotl(s[b] ^ s[c], 12)
#|     s[a] = (s[a] + s[b]) & 0xffffffff; s[d] = rotl(s[d] ^ s[a], 8)
#|     s[c] = (s[c] + s[d]) & 0xffffffff; s[b] = rotl(s[b] ^ s[c], 7)
#| def rounds(s):
#|     for _ in range(10):
#|         qr(s, 0, 4, 8, 12); qr(s, 1, 5, 9, 13); qr(s, 2, 6, 10, 14); qr(s, 3, 7, 11, 15)
#|         qr(s, 0, 5, 10, 15); qr(s, 1, 6, 11, 12); qr(s, 2, 7, 8, 13); qr(s, 3, 4, 9, 14)
#| CONST = [0x61707865, 0x3320646e, 0x79622d32, 0x6b206574]
#|
#| def hchacha20(key, n16):
#|     s = CONST + list(struct.unpack('<8I', key)) + list(struct.unpack('<4I', n16))
#|     rounds(s)
#|     return struct.pack('<8I', *(s[0:4] + s[12:16]))
#|
#| def chacha_block(key, counter, n12):        # RFC 8439 §2.3
#|     s0 = CONST + list(struct.unpack('<8I', key)) + [counter] + list(struct.unpack('<3I', n12))
#|     s = list(s0); rounds(s)
#|     return struct.pack('<16I', *[(a + b) & 0xffffffff for a, b in zip(s, s0)])
#|
#| def chacha20(key, n12, data, counter=1):
#|     out = b''
#|     for i in range(0, len(data), 64):
#|         out += bytes(a ^ b for a, b in zip(data[i:i + 64], chacha_block(key, counter + i // 64, n12)))
#|     return out
#|
#| def poly1305(key, msg):                     # RFC 8439 §2.5
#|     r = int.from_bytes(key[:16], 'little') & 0x0ffffffc0ffffffc0ffffffc0fffffff
#|     s = int.from_bytes(key[16:], 'little'); acc = 0; p = (1 << 130) - 5
#|     for i in range(0, len(msg), 16):
#|         n = int.from_bytes(msg[i:i + 16] + b'\x01', 'little')
#|         acc = (acc + n) * r % p
#|     return ((acc + s) & ((1 << 128) - 1)).to_bytes(16, 'little')
#|
#| def pad16(b): return b + bytes((16 - len(b) % 16) % 16)
#|
#| def aead_ref(key, n12, ad, plain):          # RFC 8439 §2.8
#|     otk = chacha_block(key, 0, n12)[:32]
#|     cipher = chacha20(key, n12, plain)
#|     tag = poly1305(otk, pad16(ad) + pad16(cipher) + struct.pack('<QQ', len(ad), len(cipher)))
#|     return cipher + tag
#|
#| def xchacha_seal(key, nonce, ad, plain):
#|     sub = hchacha20(key, nonce[:16]); n12 = b'\0' * 4 + nonce[16:]
#|     a = ChaCha20Poly1305(sub).encrypt(n12, plain, ad)
#|     b = aead_ref(sub, n12, ad, plain)
#|     assert a == b and len(a) == 48
#|     return a
#|
#|
#| def ctx(cid, gen): return cid + u48(gen)
#|
#| def key_for_gen(seed, cid, gen, rnd):     # Hedge(issuer_sk, "kome/key", cid || u48(gen), rnd)
#|     assert len(seed) == 32 and len(cid) == 32 and len(rnd) == 32
#|     return f(seed, b'kome/key' + ctx(cid, gen) + rnd)

#| def eph(kg, cid, gen, signer):
#|     i = b'kome/eph' + ctx(cid, gen) + signer
#|     assert len(i) == 78
#|     return f(kg, i)
#|
#| def box(kg, cid, gen, signer, dh_pk, sign_pk):
#|     eph_sk = eph(kg, cid, gen, signer)
#|     box_pk = x25519_pub(eph_sk)
#|     wi = b'kome/wrap' + box_pk + dh_pk + sign_pk + ctx(cid, gen)
#|     assert len(wi) == 143
#|     wrap = f(x25519(eph_sk, dh_pk), wi)
#|     nonce = f(kg, b'kome/box' + dh_pk)[:24]
#|     return eph_sk, box_pk, wrap, nonce, xchacha_seal(wrap, nonce, ctx(cid, gen), kg)
